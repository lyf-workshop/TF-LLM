async def _try_generate_l1(self, step: int):
    # 获取未聚合的 L0
    recent_l0 = self._get_unaggregated_l0()
    
    # 检查阈值
    if len(recent_l0) >= self.h_config.l1_aggregation_threshold:  # 默认 5
        # 取前 5 个 L0
        l0_batch = recent_l0[:5]
        
        # ⭐ 调用 LLM 生成 L1
        l1_content = await self._generate_l1_from_l0(l0_batch)
        
        # 保存 L1
        l1_exp = {

            'content': l1_content,
            'source_l0_ids': [exp['id'] for exp in l0_batch],
            'step': step
        }
        self.l1_experiences.append(l1_exp)

async def _try_generate_l2(self, step: int):
    # 获取未聚合的 L1
    recent_l1 = self._get_unaggregated_l1()
    
    # 检查阈值
    if len(recent_l1) >= self.h_config.l2_aggregation_threshold:  # 默认 3
        # 取前 3 个 L1
        l1_batch = recent_l1[:3]
        
        # 获取所有源 L0（约 15 个）
        source_l0_ids = set()
        for l1 in l1_batch:
            source_l0_ids.update(l1['source_l0_ids'])
        source_l0 = [exp for exp in self.l0_experiences if exp['id'] in source_l0_ids]
        
        # ⭐ 调用 LLM 生成 L2（基于 L1 + 源 L0）
        l2_content = await self._generate_l2_from_l1_and_l0(l1_batch, source_l0)
        
        # 保存 L2
        l2_exp = {
            'id': f"L2_{len(self.l2_experiences)}",
            'content': l2_content,
            'source_l1_ids': [exp['id'] for exp in l1_batch],
            'step': step
        }
        self.l2_experiences.append(l2_exp)

## 🎯 关键设计亮点

### 1. **增量生成**

- ✅ 每个 epoch 后自动检查聚合条件
- ✅ 满足条件立即生成（无需等待训练结束）
- ✅ 允许训练中期就使用 L1/L2 经验

### 2. **L2 双重输入**

- ❌ **传统方法**：`L2 = LLM(L1_batch)` → 容易过度抽象
- ✅ **本系统**：`L2 = LLM(L1_batch + source_L0)` → 保持实用性

**为什么这样设计？**
- L1 已经是抽象的模式
- 仅基于 L1 再抽象 → L2 会变得泛泛而谈
- 提供源 L0 → L2 能看到具体场景，避免空洞的"大道理"

### 3. **谨慎去重**

```python
# L0 去重条件
if (
    same_scope_key  # 同一游戏/任务
    and Jaccard_similarity >= 0.95  # 几乎完全相同
):
    skip()
else:
    keep()
```

**原则**：
- ✅ 宁可保留重复，也不误杀有效变体
- ✅ 只去除几乎完全相同的经验
- ✅ 不同任务/游戏的经验从不去重

### 4. **聚合阈值可配置**

```yaml
# 可根据任务复杂度调整
l1_aggregation_threshold: 5  # 激进: 3, 保守: 10
l2_aggregation_threshold: 3  # 激进: 2, 保守: 5
```

**建议**：
- **简单任务**（Alphabetical Sorting）：阈值可以更低（L1=3, L2=2）
- **复杂任务**（逻辑推理）：阈值应更高（L1=10, L2=5）
- **Wordle**：默认值合适（L1=5, L2=3）

## 📈 数量预估

### 20 个问题，3 个 epoch

**假设**：每个问题生成 1 个 L0

```
Epoch 1: 
- L0: 10 个 (新增)
- L1: 2 个 (每 5 个 L0 → 1 个 L1)
- L2: 0 个 (需要 3 个 L1)

Epoch 2:
- L0: 20 个 (累积)
- L1: 4 个 (累积)
- L2: 1 个 (每 3 个 L1 → 1 个 L2)

Epoch 3:
- L0: 30 个 (累积)
- L1: 6 个 (累积)
- L2: 2 个 (累积)
```

**公式**：
```python
# 每个 epoch 的 L0 数量
l0_per_epoch = problem_count * max_l0_per_problem  # 默认 20 * 1 = 20

# 总 L0 数量
total_l0 = l0_per_epoch * num_epochs  # 20 * 3 = 60（去重后约 30）

# 总 L1 数量（估算）
total_l1 = total_l0 // l1_aggregation_threshold  # 30 // 5 = 6

# 总 L2 数量（估算）
total_l2 = total_l1 // l2_aggregation_threshold  # 6 // 3 = 2
```
